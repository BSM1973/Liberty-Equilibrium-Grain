cmake_minimum_required(VERSION 3.22)
project(EquilibriumGrain VERSION 0.9.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
option(EG_PLUGIN "Build JUCE plugin" ON)
include(CTest)
add_executable(grain_test Tests/GrainTest.cpp)
add_test(NAME grain_test COMMAND grain_test)
add_executable(filter_effects_test Tests/FilterEffectsTest.cpp)
add_test(NAME filter_effects_test COMMAND filter_effects_test)
add_executable(modulation_test Tests/ModulationTest.cpp)
add_test(NAME modulation_test COMMAND modulation_test)
add_executable(factory_sample_test Tests/FactorySampleTest.cpp)
add_test(NAME factory_sample_test COMMAND factory_sample_test)
add_executable(spectral_test Tests/SpectralTest.cpp)
add_test(NAME spectral_test COMMAND spectral_test)
add_executable(performance_sequencer_test Tests/PerformanceSequencerTest.cpp)
add_test(NAME performance_sequencer_test COMMAND performance_sequencer_test)
add_executable(expression_test Tests/ExpressionTest.cpp)
add_test(NAME expression_test COMMAND expression_test)
add_executable(hybrid_oscillator_test Tests/HybridOscillatorTest.cpp)
add_test(NAME hybrid_oscillator_test COMMAND hybrid_oscillator_test)
add_executable(sample_tools_test Tests/SampleToolsTest.cpp)
add_test(NAME sample_tools_test COMMAND sample_tools_test)
add_executable(quality_control_test Tests/QualityControlTest.cpp)
add_test(NAME quality_control_test COMMAND quality_control_test)
set_tests_properties(quality_control_test PROPERTIES TIMEOUT 900)
if(EG_PLUGIN)
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
  set(formats VST3 Standalone)
  if(APPLE)
    list(APPEND formats AU)
  endif()
  juce_add_plugin(EquilibriumGrain COMPANY_NAME "Liberty Music Software" BUNDLE_ID com.libertymusicsoftware.equilibriumgrain PLUGIN_MANUFACTURER_CODE Lbty PLUGIN_CODE EqGr FORMATS ${formats} PRODUCT_NAME "Equilibrium Grain" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(EquilibriumGrain PRIVATE Source/Plugin.cpp)
  target_compile_definitions(EquilibriumGrain PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(EquilibriumGrain PRIVATE juce::juce_audio_utils juce::juce_audio_processors juce::juce_gui_extra PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
  juce_add_console_app(eg_plugin_quality PRODUCT_NAME "Equilibrium Grain Quality")
  target_sources(eg_plugin_quality PRIVATE Tests/PluginQualityTest.cpp)
  target_compile_definitions(eg_plugin_quality PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(eg_plugin_quality PRIVATE juce::juce_audio_utils juce::juce_audio_processors juce::juce_gui_extra juce::juce_recommended_config_flags)
  add_test(NAME plugin_quality COMMAND eg_plugin_quality)
  set_tests_properties(plugin_quality PROPERTIES TIMEOUT 900)
  if(MSVC)
    target_compile_options(EquilibriumGrain PRIVATE /utf-8)
    target_compile_options(eg_plugin_quality PRIVATE /utf-8)
  endif()
endif()
